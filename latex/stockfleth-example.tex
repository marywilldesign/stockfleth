\section*{Code}
Inline: \texttt{if (x != 0) return 1l;} — zero is slashed, and 1 l i I are distinct.

\begin{Verbatim}[fontsize=\small]
def greet(name: str) -> str:
    # fixed cells keep every column aligned
    return f"hei, {name}! 0O 1lI |"
\end{Verbatim}
